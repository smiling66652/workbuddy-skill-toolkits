# 微信小程序云开发指南

## 初始化云环境 #

### app.js #

```javascript
// app.js
App({
  onLaunch() {
    // 初始化云开发环境
    wx.cloud.init({
      env: 'your-env-id' // 云开发环境 ID
    });
  }
})
```

## 云函数 #

### 调用云函数 #

```javascript
// pages/index/index.js
Page({
  async callCloudFunction() {
    try {
      const res = await wx.cloud.callFunction({
        name: 'addTodo', // 云函数名称
        data: {
          title: '学习微信小程序',
          done: false
        }
      });
      console.log(res.result);
    } catch (err) {
      console.error(err);
    }
  }
})
```

### 云函数示例 #

```javascript
// cloudfunctions/addTodo/index.js
const cloud = require('wx-server-sdk');
cloud.init({
  env: cloud.DYNAMIC_CURRENT_ENV
});

exports.main = async (event, context) => {
  const { OPENID, APPID } = cloud.getWXContext(); // 获取 OPENID
  const db = cloud.database();
  
  await db.collection('todos').add({
    data: {
      ...event,
      _openid: OPENID,
      createTime: db.serverDate()
    }
  });
  
  return { success: true };
}
```

## 云数据库 #

### 插入数据 #

```javascript
// pages/index/index.js
Page({
  async addData() {
    const db = wx.cloud.database();
    try {
      const res = await db.collection('todos').add({
        data: {
          title: '学习微信小程序',
          done: false
        }
      });
      console.log(res._id);
    } catch (err) {
      console.error(err);
    }
  }
})
```

### 查询数据 #

```javascript
// pages/index/index.js
Page({
  async getData() {
    const db = wx.cloud.database();
    try {
      const res = await db.collection('todos').where({
        done: false
      }).get();
      console.log(res.data);
    } catch (err) {
      console.error(err);
    }
  }
})
```

### 更新数据 #

```javascript
// pages/index/index.js
Page({
  async updateData() {
    const db = wx.cloud.database();
    try {
      await db.collection('todos').doc('todo-id').update({
        data: {
          done: true
        }
      });
    } catch (err) {
      console.error(err);
    }
  }
})
```

### 删除数据 #

```javascript
// pages/index/index.js
Page({
  async deleteData() {
    const db = wx.cloud.database();
    try {
      await db.collection('todos').doc('todo-id').remove();
    } catch (err) {
      console.error(err);
    }
  }
})
```

## 云存储 #

### 上传文件 #

```javascript
// pages/index/index.js
Page({
  async uploadFile() {
    try {
      const res = await wx.cloud.uploadFile({
        cloudPath: 'example.png', // 云存储路径
        filePath: '/local/path/example.png' // 本地文件路径
      });
      console.log(res.fileID); // 云文件 ID
    } catch (err) {
      console.error(err);
    }
  }
})
```

### 下载文件 #

```javascript
// pages/index/index.js
Page({
  async downloadFile() {
    try {
      const res = await wx.cloud.downloadFile({
        fileID: 'cloud://example.png' // 云文件 ID
      });
      console.log(res.tempFilePath); // 临时文件路径
    } catch (err) {
      console.error(err);
    }
  }
})
```

### 获取临时链接 #

```javascript
// pages/index/index.js
Page({
  async getTempFileURL() {
    try {
      const res = await wx.cloud.getTempFileURL({
        fileList: ['cloud://example.png']
      });
      console.log(res.fileList[0].tempFileURL); // 临时链接
    } catch (err) {
      console.error(err);
    }
  }
})
```

## 用户登录 #

### 获取 OPENID #

```javascript
// cloudfunctions/getOpenId/index.js
const cloud = require('wx-server-sdk');
cloud.init({
  env: cloud.DYNAMIC_CURRENT_ENV
});

exports.main = async (event, context) => {
  const { OPENID, APPID } = cloud.getWXContext(); // 获取 OPENID
  return { openid: OPENID };
}
```

## 故障排查 #

| 问题 | 解决方案 |
|------|-----------|
| 云函数调用失败 | 检查云函数日志（CloudBase 控制台 → 云函数 → 日志） |
| 云数据库权限错误 | 检查安全规则（CloudBase 控制台 → 数据库 → 权限设置） |
| 云存储上传失败 | 检查文件大小限制（最大 50MB） |
| 用户登录失败 | 检查 AppID 和 AppSecret 是否配置正确 |
