# 微信小程序框架指南

## 项目结构 #

```
├── app.js          # 小程序逻辑
├── app.json        # 小程序配置
├── app.wxss        # 全局样式
├── pages/           # 页面目录
│   └── index/
│       ├── index.js    # 页面逻辑
│       ├── index.json  # 页面配置
│       ├── index.wxml  # 页面模板
│       └── index.wxss  # 页面样式
└── components/     # 自定义组件目录
```

---

## 应用生命周期 #

```javascript
// app.js
App({
  onLaunch(options) {
    // 小程序初始化（全局只触发一次）
    console.log('小程序启动', options);
  },
  onShow(options) {
    // 小程序从前台进入前台
    console.log('小程序显示', options);
  },
  onHide() {
    // 小程序从前台进入后台
    console.log('小程序隐藏');
  },
  globalData: {
    userInfo: null
  }
})
```

## 页面生命周期 #

```javascript
// pages/index/index.js
Page({
  data: {
    message: 'Hello World'
  },
  onLoad(options) {
    // 页面加载（一个页面只调用一次）
    console.log('页面加载', options);
  },
  onShow() {
    // 页面显示（每次打开页面都调用）
    console.log('页面显示');
  },
  onReady() {
    // 页面初次渲染完成（一个页面只调用一次）
    console.log('页面就绪');
  },
  onHide() {
    // 页面隐藏（进入后台或跳转到其他页面）
    console.log('页面隐藏');
  },
  onUnload() {
    // 页面卸载（redirectTo 或 navigateBack）
    console.log('页面卸载');
  }
})
```

## 自定义组件 #

### 创建组件

```
components/
└── my-component/
    ├── my-component.js    # 组件逻辑
    ├── my-component.json  # 组件配置
    ├── my-component.wxml  # 组件模板
    └── my-component.wxss  # 组件样式
```

### 组件配置

```json
// components/my-component/my-component.json
{
  "component": true,
  "usingComponents": {}
}
```

### 组件逻辑

```javascript
// components/my-component/my-component.js
Component({
  properties: {
    // 父 → 子（属性）
    prop: {
      type: String,
      value: 'default value'
    }
  },
  data: {
    // 组件内部数据
    internalData: 'internal'
  },
  methods: {
    // 组件方法
    customMethod() {
      this.setData({
        internalData: 'updated'
      });
    }
  },
  lifetimes: {
    // 组件生命周期
    attached() {
      console.log('组件 attached');
    },
    detached() {
      console.log('组件 detached');
    }
  }
})
```

### 使用组件

```json
// pages/index/index.json
{
  "usingComponents": {
    "my-component": "/components/my-component/my-component"
  }
}
```

```html
<!-- pages/index/index.wxml -->
<my-component prop="value" />
```

## 常用 API #

### 网络请求

```javascript
// pages/index/index.js
Page({
  fetchData() {
    wx.request({
      url: 'https://api.example.com/data',
      method: 'GET',
      data: {
        page: 1,
        limit: 10
      },
      success: (res) => {
        console.log(res.data);
      },
      fail: (err) => {
        console.error(err);
      }
    });
  }
})
```

### 本地存储

```javascript
// 存储数据
wx.setStorageSync('key', 'value');

// 读取数据
const value = wx.getStorageSync('key');

// 删除数据
wx.removeStorageSync('key');

// 清除所有数据
wx.clearStorageSync();
```

### 图片预览

```javascript
// pages/index/index.js
Page({
  previewImage() {
    wx.previewImage({
      current: 'https://img.example.com/1.jpg', // 当前显示图片
      urls: [ // 所有图片 URL 列表
        'https://img.example.com/1.jpg',
        'https://img.example.com/2.jpg',
        'https://img.example.com/3.jpg'
      ]
    });
  }
})
```

### 扫码

```javascript
// pages/index/index.js
Page({
  scanCode() {
    wx.scanCode({
      success: (res) => {
        console.log(res.result); // 扫码结果
      }
    });
  }
})
```

## 故障排查 #

| 问题 | 解决方案 |
|------|-----------|
| 页面不显示 | 检查 `app.json` 的 `pages` 字段是否配置了页面路径 |
| 数据不更新 | 检查是否使用 `this.setData({})` 更新数据 |
| 事件不触发 | 检查事件绑定是否正确（`bindtap` vs `catchtap`） |
| 组件不显示 | 检查 `json` 中的 `usingComponents` 是否配置正确 |
| API 调用失败 | 检查 API 是否需要配置域名白名单（微信公众平台 → 开发管理 → 开发设置 → 服务器域名） |
